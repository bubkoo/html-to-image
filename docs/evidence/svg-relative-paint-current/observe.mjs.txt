import { readFile, writeFile, mkdir } from 'node:fs/promises'
import { createServer } from 'node:http'
import { resolve, extname } from 'node:path'
import { pathToFileURL } from 'node:url'
const { chromium, firefox, webkit } = await import(pathToFileURL(resolve(process.env.HTI_REPO_ROOT || process.cwd(), 'node_modules/@playwright/test/index.mjs')).href)

const root = process.cwd()
const out = resolve(root, 'output/playwright/svg-redirect-current')
await mkdir(out, { recursive: true })
const svg = (color, name) => `<svg xmlns="http://www.w3.org/2000/svg" width="20" height="20"><defs><linearGradient id="paint"><stop stop-color="${color}"/></linearGradient></defs><symbol id="icon" viewBox="0 0 20 20"><rect width="20" height="20" fill="url(${name}#paint)"/></symbol></svg>`
const flat = '<svg xmlns="http://www.w3.org/2000/svg"><symbol id="icon" viewBox="0 0 20 20"><rect width="20" height="20" fill="green"/></symbol></svg>'
const requests = []
const server = createServer(async (req, res) => {
  const u = new URL(req.url, 'http://localhost')
  requests.push(u.pathname)
  if (u.pathname === '/probe/') {
    res.writeHead(200, { 'Content-Type': 'text/html', 'Cache-Control': 'no-store' }).end('<!doctype html><meta name="viewport" content="width=device-width,initial-scale=1"><body style="margin:0">')
    return
  }
  const m = /^\/probe\/([a-f0-9-]{36})\/(.*)$/.exec(u.pathname)
  if (m) {
    const path = m[2], prefix = '/probe/' + m[1] + '/'
    if (path === 'redirect.svg') {
      res.writeHead(302, { Location: prefix + 'assets/sprite.svg', 'Cache-Control': 'no-store' }).end()
      return
    }
    const entry = { 'assets/sprite.svg': ['green', 'sprite.svg'], 'sprite.svg': ['red', 'sprite.svg'], 'direct.svg': ['green', 'direct.svg'] }[path]
    if (entry || path === 'flat.svg') {
      res.writeHead(200, { 'Content-Type': 'image/svg+xml', 'Cache-Control': 'no-store' }).end(entry ? svg(...entry) : flat)
      return
    }
  }
  const p = resolve(root, '.' + u.pathname)
  if (!u.pathname.startsWith('/es/') || !p.startsWith(root + '/es/')) { res.writeHead(404).end(); return }
  try { res.writeHead(200, { 'Content-Type': extname(p) === '.js' ? 'text/javascript' : 'text/plain' }).end(await readFile(p)) }
  catch { res.writeHead(404).end() }
})
await new Promise(resolve => server.listen(4493, '127.0.0.1', resolve))
const createScene = new Function('return (' + await readFile('docs/evidence/svg-redirect-create-scene.js.txt', 'utf8') + ')')()
const captureScene = async () => {
  const {createRenderer} = await import('/es/index.js')
  const {root,source} = window.__scene
  const requests=[]
  const renderer=createRenderer({pixelRatio:1,skipFonts:true,resources:{resolve:({url})=>{requests.push(url)}}})
  let capture
  try {
    capture=await renderer.capture(root)
    const dataURL=await capture.toPng()
    const image=await capture.toImageData()
    return {dataURL,pixels:[10,30,50,70].map(x=>Array.from(image.data.slice(4*(10*image.width+x),4*(10*image.width+x)+4))),sourceUnchanged:source===root.outerHTML,requests,userAgent:navigator.userAgent,visibility:document.visibilityState}
  } finally {capture?.dispose();renderer.dispose()}
}
const results = {}
try {
  for (const [name, engine] of Object.entries({ chromium, firefox, webkit })) {
    const browser = await engine.launch()
    try {
      const page = await browser.newPage({ deviceScaleFactor: 1 })
      await page.goto('http://127.0.0.1:4493/probe/')
      const nativeStart = requests.length
      const input = await page.evaluate(createScene)
      await page.waitForFunction(() => Array.from(document.querySelectorAll('use')).every(x => x.getBBox().width === 20))
      await page.waitForLoadState('networkidle')
      const source = page.locator('#source')
      await source.screenshot({ path: resolve(out, name + '-before.png'), scale: 'css' })
      const start = requests.length
      const hit = await page.evaluate(() => [10,30,50,70].map(x => {const n=document.elementFromPoint(x,10); return {tag:n?.localName,href:n?.getAttribute('href')}}))
      const result = {...await page.evaluate(captureScene), hit}
      await source.screenshot({ path: resolve(out, name + '-after.png'), scale: 'css' })
      await writeFile(resolve(out, name + '-capture.png'), Buffer.from(result.dataURL.split(',')[1], 'base64'))
      delete result.dataURL
      results[name] = { input, result, nativeHTTPRequests: requests.slice(nativeStart, start), captureHTTPRequests: requests.slice(start) }
      await writeFile(resolve(out, name + '.json'), JSON.stringify(results[name], null, 2))
      process.stdout.write(JSON.stringify({ name, ...result }) + '\n')
    } finally { await browser.close() }
  }
} finally { await new Promise(resolve => server.close(resolve)) }
await writeFile(resolve(out, 'summary.json'), JSON.stringify(results, null, 2))
